31def write_tfs( 

32 tfs_file_path: pathlib.Path | str, 

33 data_frame: TfsDataFrame | pd.DataFrame | pd.Series, 

34 headers_dict: dict | None = None, 

35 save_index: str | bool = False, # noqa: FBT002 

36 colwidth: int = DEFAULT_COLUMN_WIDTH, 

37 headerswidth: int = DEFAULT_COLUMN_WIDTH, 

38 non_unique_behavior: str = "warn", 

39 validate: str | None = None, 

40) -> None: 

41 """ 

42 Writes the provided `DataFrame` to disk at **tfs_file_path**. If `headers_dict` 

43 is provided it is written to dist as the headers. Note that this function is 

44 exported also at the top-level of the package as `tfs.write`. 

45 

46 .. note:: 

47 Compression of the output file is possible, by simply providing a valid compression extension 

48 as the **tfs_file_path** suffix. Any compression format supported by ``pandas`` is accepted, 

49 which includes: ``.gz``, ``.bz2``, ``.zip``, ``.xz``, ``.zst``, ``.tar``, ``.tar.gz``, 

50 ``.tar.xz`` or ``.tar.bz2``. See below for examples. 

51 

52 .. warning:: 

53 Through the *validate* argument, one can skip dataframe validation before writing 

54 to file which can improve performance. This is **not recommended** if the file needs 

55 to be read by ``MAD-X`` or ``MAD-NG``. The default behaviour is to check for ``MAD-X`` 

56 compatibility. The option to skip validation is left for the user to use (at their own 

57 risk) should they wish to avoid lengthy validation of large `TfsDataFrames` (such as 

58 for instance a sliced FCC lattice). 

59 

60 Args: 

61 tfs_file_path (pathlib.Path | str): Path to the output **TFS** file. 

62 data_frame (TfsDataFrame | pd.DataFrame | pd.Series): The dataframe to write to file. If 

63 a Series-like object is given, it will be converted to a `TfsDataFrame` first and 

64 written with a single column. 

65 headers_dict (dict): Headers for the `data_frame`. If not provided, assumes a 

66 `TfsDataFrame` was given and tries to use ``data_frame.headers``. Writes with 

67 empty headers is those are not found either. 

68 save_index (str | bool): bool or string. Default to ``False``. If ``True``, saves 

69 the index of `data_frame` to a column identifiable by `INDEX&&&`. If given as string, 

70 saves the index of `data_frame` to a column named by the provided value. 

71 colwidth (int): Column width, can not be smaller than `MIN_COLUMN_WIDTH`. 

72 headerswidth (int): Used to format the header width for both keys and values. 

73 non_unique_behavior (str): behavior to adopt if non-unique indices or columns are found in the 

74 dataframe. Accepts `warn` and `raise` as values, case-insensitively, which dictates 

75 to respectively issue a warning or raise an error if non-unique elements are found. 

76 validate (str): Determines if and which validation will be performed before writing. By 

77 default no validation is performed. Accepted values are `madx`, `mad-x`, `madng` and 

78 `mad-ng` (case-insensitive), for compatibility with ``MAD-X`` and ``MAD-NG`` codes, 

79 respectively. See the `tfs.frame.validate` function for more information on the 

80 validation steps. 

81 

82 Examples: 

83 Writing to file is simple, as most arguments have sane default values. 

84 The simplest usage goes as follows: 

85 

86 .. code-block:: python 

87 

88 tfs.write("filename.tfs", dataframe) 

89 

90 One can choose to perform dataframe validation before writing 

91 it to file. This can be done by providing an accepted compatibility 

92 mode to check for (either `madx` or `madng`), as: 

93 

94 .. code-block:: python 

95 

96 tfs.write("filename.tfs", dataframe, validate="madx") 

97 

98 If one wants to, for instance, raise and error on non-unique indices or 

99 columns when validating the dataframe, one can do so as: 

100 

101 .. code-block:: python 

102 

103 tfs.write( 

104 "filename.tfs", dataframe, non_unique_behavior="raise", validate="madng" 

105 ) 

106 

107 It is possible to directly have the output file be compressed, by specifying a 

108 valid compression extension as the **tfs_file_path** suffix. The detection 

109 and compression is handled automatically. For instance: 

110 

111 .. code-block:: python 

112 

113 tfs.write("filename.tfs.gz", dataframe) 

114 """ 

115 left_align_first_column = False 

116 tfs_file_path = pathlib.Path(tfs_file_path) 

117 

118 # Check for provided headers. If None, try to get them from the df 

119 # and default to empty dict (will not write any lines to file) 

120 if headers_dict is None: 

121 headers_dict = getattr(data_frame, "headers", {}) 

122 

123 # We case to a TfsDataFrame anyway for validation and writing 

124 # We ensure column names as strings in case it was a pd.Series-like (and colname is 0) 

125 data_frame = TfsDataFrame(data_frame, headers=headers_dict) 

126 data_frame.columns = data_frame.columns.astype(str) 

127 

128 # Only perform validation if asked (validation is OFF by default) 

129 # We also check for False as it was the way to skip it in tfs-pandas 3.x 

130 if validate is not None and validate is not False: # validation function will check for valid values 

131 validate_frame( 

132 data_frame, 

133 info_str=f"to be written in {tfs_file_path.absolute()}", 

134 non_unique_behavior=non_unique_behavior, 

135 compatibility=validate, 

136 ) 

137 

138 # Let pandas try to infer the best dtypes for the data to write (only to write, the 

139 # actual dataframe provided by the user is not changed so this operation is fine). 

140 # Passed options: don't convert float to ints, don't try (and fail) to convert complex 

141 # to floats, and don't try to convert strings to its special StringDtype (as then 'None' 

142 # would be transformed into <pd.NA> and if we write this to file we are very much cooked.) 

143 # Overall we do not care to infer specialized dtypes, just that it makes the best inference 

144 # to valid dtypes (i.e. an object column should be inferred as strings if that makes sense). 

145 data_frame = data_frame.convert_dtypes(convert_integer=False, convert_floating=False, convert_string=False) 

146 

147 if save_index: 

148 left_align_first_column = True 

149 _insert_index_column(data_frame, save_index) 

150 

151 colwidth = max(MIN_COLUMN_WIDTH, colwidth) 

152 headers_str = _get_headers_string(headers_dict, headerswidth) 

153 colnames_str = _get_colnames_string(data_frame.columns, colwidth, left_align_first_column) 

154 coltypes_str = _get_coltypes_string(data_frame.dtypes, colwidth, left_align_first_column) 

155 data_str = _get_data_string(data_frame, colwidth, left_align_first_column) 

156 

157 LOGGER.debug(f"Attempting to write file: {tfs_file_path.name} in {tfs_file_path.parent}") 

158 with get_handle(tfs_file_path, mode="w", compression="infer") as output_path: 

159 tfs_handle = output_path.handle 

160 tfs_handle.write( # the last "\n" is to have an EOL at EOF, which is UNIX standard 

161 "\n".join(line for line in (headers_str, colnames_str, coltypes_str, data_str) if line) + "\n" 

162 )

168def _insert_index_column( 

169 data_frame: TfsDataFrame | pd.DataFrame, save_index: str | bool | None = None 

170) -> None: 

171 """ 

172 Inserts the index of the dataframe into it as a column, naming it according to 

173 'save_index' if it was provided. Otherwise it tries to use the existing index's 

174 name (if present) and falls back to a default. 

175 """ 

176 if isinstance(save_index, str): # save index into column by name given 

177 idx_name = save_index 

178 else: # save index into column, which can be found by INDEX_ID 

179 try: 

180 idx_name = INDEX_ID + data_frame.index.name 

181 except TypeError: 

182 idx_name = INDEX_ID 

183 data_frame.insert(0, idx_name, data_frame.index)

186def _get_headers_string(headers_dict: dict, width: int) -> str: 

187 """ 

188 Returns the string to write a ``TfsDataFrame`` headers to file. Will return an empty string if 

189 called for an empty headers dictionary, in order not write a line to file. 

190 

191 Args: 

192 headers_dict (dict): the ``TfsDataFrame`` headers. 

193 width (int): column width to use when formatting keys and values from the headers dict. 

194 

195 Returns: 

196 A full string representation for the headers dictionary, TFS compliant. 

197 """ 

198 if headers_dict: 

199 return "\n".join(_get_header_line(name, value, width) for name, value in headers_dict.items()) 

200 return ""

203def _get_header_line(name: str, value, width: int) -> str: 

204 """ 

205 Creates and returns the string value for a single header line, based 

206 on the name of the header parameter and its value. 

207 

208 For instance, calling this for 'param' header which is a float equal to 

209 1.792 and using the DEFAULT_COLUMN_WIDTH of the package would yield: 

210 "@ param %le 1.792" 

211 

212 

213 Args: 

214 name (str): name of the header parameter. 

215 value: value of the header parameter. Any valid type is accepted here 

216 (int, float, str, potentially bool, complex etc.) and the type 

217 of this value is used to infer the formatting. 

218 width (int): column width to use when formatting the header line. 

219 

220 Returns: 

221 The full, formatted header line string. 

222 """ 

223 if not isinstance(name, str): 

224 errmsg = f"{name} is not a string" 

225 raise TypeError(errmsg) 

226 type_identifier = _value_to_tfs_type_identifier(value) 

227 dtype_ = ( 

228 NoneType if value is None else np.array(value).dtype 

229 ) # otherwise numpy gives 'Object' for 'None's 

230 # Strip the following as it might have trailing spaces and we leave that to the alignment formatting below 

231 value_str = ( 

232 ValueToStringFormatter().format_field(value, _dtype_to_formatter_string(dtype_, width)).strip() 

233 ) 

234 return f"@ {name:<{width}} {type_identifier} {value_str.strip():>{width}}"

237def _get_colnames_string(colnames: list[str], colwidth: int, left_align_first_column: bool) -> str: # noqa: FBT001 

238 """Returns the string for the line with the column names.""" 

239 format_string = _get_row_format_string([None] * len(colnames), colwidth, left_align_first_column) 

240 return "* " + format_string.format(*colnames)

243def _get_coltypes_string(types: pd.Series, colwidth: int, left_align_first_column: bool) -> str: # noqa: FBT001 

244 """Returns the string for the line with the column type specifiers.""" 

245 fmt = _get_row_format_string([str] * len(types), colwidth, left_align_first_column)

249def _get_data_string( 

250 data_frame: TfsDataFrame | pd.DataFrame, 

251 colwidth: int, 

252 left_align_first_column: bool, # noqa: FBT001 

253) -> str: 

254 """ 

255 Returns the complete string to be written for the data part of the dataframe. 

256 This corresponds to all the data rows, after the column names and the column 

257 type specifiers. 

258 

259 Args: 

260 data_frame (TfsDataFrame | pd.DataFrame): the dataframe to write. 

261 colwidth (int): column width to use when formatting the data. 

262 left_align_first_column (bool): whether to left-align the first column or not. 

263 

264 Returns: 

265 The full string representation of the data part of the dataframe. 

266 """ 

267 if len(data_frame.index) == 0 or len(data_frame.columns) == 0: 

268 return "\n" 

269 

270 format_strings = " " + _get_row_format_string(data_frame.dtypes, colwidth, left_align_first_column) 

271 data_frame = data_frame.astype(object) # overrides pandas auto-conversion (lead to format bug) 

272 string_formatter = ValueToStringFormatter() 

273 

274 # A bit ugly but we go line per line and determine the string on the fly 

275 return "\n".join( 

276 data_frame.apply(lambda series: string_formatter.format(format_strings, *series), axis=1) 

277 )

280def _get_row_format_string( 

281 dtypes: list[type], 

282 colwidth: int, 

283 left_align_first_column: bool, # noqa: FBT001 

284) -> str: 

285 """ 

286 Returns the formatter string for a given row of the data part of the dataframe, 

287 based on the dtypes of the columns and the column width to use for writing. It is 

288 a string with the formatting speficiers (for fstrings), one slot per column. For 

289 instance: {0:>20s} {1:>20.12g} {2:>20d} {3:>20.12g}". 

290 

291 Args: 

292 dtypes (list): list of the dtypes of the columns. 

293 colwidth (int): column width to use when formatting the row. 

294 left_align_first_column (bool): whether to left-align the first column or not. 

295 

296 Returns: 

297 The full formatter string for any data row. 

298 """ 

299 return " ".join( 

300 f"{{{indx:d}:" 

301 f"{'<' if (not indx) and left_align_first_column else '>'}" 

302 f"{_dtype_to_formatter_string(type_, colwidth)}}}" 

303 for indx, type_ in enumerate(dtypes) 

304 )

307def _value_to_tfs_type_identifier(value) -> str: 

308 """ 

309 Returns the **TFS** dtype identifier for the provided value, 

310 as a string. For instance for a float, it would return "%le". 

311 """ 

312 # First intercept for None values (MAD-NG writes nil) to return %n 

313 # (as the inferred dtype is object and that would be a mess later) 

314 if value is None: 

315 return "%n" 

316 

317 # Otherwise we infer the dtype and return the corresponding identifier 

318 dtype_ = np.array(value).dtype # let numpy handle conversion to it dtypes 

319 return _dtype_to_tfs_format_identifier(dtype_)

322def _dtype_to_tfs_format_identifier(type_: type) -> str: 

323 """ 

324 Return the proper **TFS** identifier for the provided dtype. This is 

325 the function called behind the scenes by `_value_to_type_identifier`, 

326 but it takes the inferred dtype as argument. For a float dtype it 

327 would return '%le', for a string '%s' etc. 

328 

329 Args: 

330 type_ (type): an instance of the built-in type (in this package, one of ``numpy`` or ``pandas`` 

331 types) to get the ID string for. 

332 

333 Returns: 

334 The ID string. 

335 

336 Raises: 

337 TypeError: if the provided type could not be identified as a valid dtype. 

338 """ 

339 if pdtypes.is_integer_dtype(type_): 

340 return "%d" 

341 if pdtypes.is_float_dtype(type_): 

342 return "%le" 

343 if pdtypes.is_string_dtype(type_): 

344 return "%s" 

345 if pdtypes.is_bool_dtype(type_): 

346 return "%b" 

347 if pdtypes.is_complex_dtype(type_): 

348 return "%lz" 

349 errmsg = f"Provided type '{type_}' could not be identified as either a bool, int, float complex or string dtype" 

350 raise TypeError(errmsg)

353def _dtype_to_formatter_string(type_: type, colsize: int) -> str: 

354 """ 

355 Return the proper formatter string for the provided dtype. This is 

356 the function called behind the scenes by `_value_to_string_format_id`, 

357 but it takes the inferred dtype as argument. For a float dtype it 

358 would return 'g', for a complex 'c' etc. It is used later for string 

359 formatting (you know, when you do f'{variable:.2f}' etc). 

360 

361 Args: 

362 type_ (type): an instance of the built-in type (in this package, one of 

363 ``numpy`` or ``pandas`` types) to get the formatter for. 

364 colsize (int): size of the written column to use for the formatter. 

365 

366 Returns: 

367 The formatter string for the provided dtype. 

368 """ 

369 type_id = _dtype_to_python_string_formatter(type_) # for Python: 'd' or 'g' or 's' etc 

370 if pdtypes.is_float_dtype(type_) or pdtypes.is_complex_dtype(type_): 

371 return f"{colsize}.{colsize - len('-0.e-000')}{type_id}" # make sure we don't round and lose info 

372 return f"{colsize}{type_id}"

375def _dtype_to_python_string_formatter(type_: type) -> str: 

376 """ 

377 Return the (Python) string-formatter for the provided dtype, 

378 a.k.a the formatter passed to f-strings ('d' for integers for 

379 instance). 

380 

381 Of special note are here "b" for boolean and "c" for complex 

382 numbers, but those are only for our own internal formatter. 

383 

384 Args: 

385 type_ (type): an instance of the built-in type (in this package, one of 

386 ``numpy`` or ``pandas`` types) to get the formatter for. 

387 

388 Returns: 

389 str: the formatter type-identifier. 

390 

391 Raises: 

392 TypeError: if the provided type could not be identified as a valid dtype. 

393 """ 

394 

395 if type_ is None: # don't know when that triggers 

396 return "" 

397 if type_ is NoneType: # this means the value was None 

398 return "s" # formatted as string since we will write 'nil' 

399 if pdtypes.is_integer_dtype(type_): 

400 return "d" 

401 if pdtypes.is_bool_dtype(type_): 

402 return "b" # can only be used with TfsStringFormatter 

403 if pdtypes.is_float_dtype(type_): 

404 return "g" 

405 if pdtypes.is_string_dtype(type_): 

406 return "s" 

407 if pdtypes.is_complex_dtype(type_): 

408 return "c" # can only be used with TfsStringFormatter 

409 

410 errmsg = f"Provided type '{type_}' could not be identified as either a bool, int, float, complex or string dtype" 

411 raise TypeError(errmsg)

417class ValueToStringFormatter(string.Formatter): 

418 """ 

419 Formatter class to be called for proper formatting of values 

420 (headers, dataframe data) into strings to write to file. 

421 """ 

422 

423 def format_field(self, value, format_spec): 

424 if format_spec.endswith("b"): # value is a boolean 

425 return self._format_boolean(value, format_spec) 

426 

427 if format_spec.endswith("c"): # value is a complex number 

428 return self._format_complex(value, format_spec) 

429 

430 if format_spec.endswith("s"): # value is a string or None 

431 return self._format_string(value, format_spec) 

432 

433 return super().format_field(value, format_spec) 

434 

435 def _format_boolean(self, value, format_spec: str): 

436 """ 

437 Special case for booleans, as from their written version 

438 will always be either 'true' or 'false' (lowercase) as 

439 done by MAD-NG. 

440 """ 

441 bool_str = str(bool(value)).lower() 

442 return super().format_field(bool_str, f"{format_spec[:-1]}s") 

443 

444 def _format_complex(self, value, format_spec: str): 

445 """ 

446 Special case for complex numbers, MAD-X and MAD-NG use 'i' for 

447 the imaginary part and not 'j', so we have to replace that in. 

448 """ 

449 return super().format_field(value, f"{format_spec[:-1]}g").replace("j", "i") 

450 

451 def _format_string(self, value: str | pathlib.Path, format_spec: str) -> str: 

452 """ 

453 Special case as we need to ensure that strings are enclosed 

454 in either " or ' quotes. This also handles `pathlib.Path` 

455 objects and makes sure they are converted to strings first. 

456 """ 

457 # First a check so that passing a Path works (i.e. if we 

458 # have a Path in the headers it will be written as a string) 

459 if isinstance(value, pathlib.Path): 

460 value = str(value) # convert its current form (can be relative) 

461 

462 # Now we go on with the formatting 

463 try: 

464 if not value.startswith(('"', "'")): 

465 value = f'"{value}"' 

466 except AttributeError: 

467 pass 

468 

469 # Special case for None values which we always write as 'nil'. In case 

470 # value is None we end up here as we want 'nil' and format it as string 

471 # We do this here as we want to write nil and not "nil" 

472 if value is None: 

473 value = "nil" 

474 

475 return super().format_field(value, format_spec)
